def make_encoder():
    try:
        return OneHotEncoder(
            handle_unknown="ignore",
            sparse_output=False
        )
    except TypeError:
        return OneHotEncoder(
            handle_unknown="ignore",
            sparse=False
        )

def make_processor(columns=FEATURES):
    cats = [c for c in CATEGORICAL if c in columns]
    nums = [c for c in columns if c not in cats]

    return ColumnTransformer(
        transformers=[
            ("categorical", make_encoder(), cats),
            ("numerical", StandardScaler(), nums)
        ],
        sparse_threshold=0
    )

def prepare_features(train_df, other_df, columns=FEATURES):
    processor = make_processor(columns)

    Xtrain = processor.fit_transform(
        train_df[columns]
    )
    Xother = processor.transform(
        other_df[columns]
    )

    return (
        processor,
        np.asarray(Xtrain, dtype=np.float32),
        np.asarray(Xother, dtype=np.float32)
    )

def make_y_scaler(y, method="standard"):
    scaler = (
        StandardScaler()
        if method == "standard"
        else MinMaxScaler()
    )

    scaler.fit(
        np.asarray(y).reshape(-1, 1)
    )

    return scaler

def scale_target(scaler, y):
    return scaler.transform(
        np.asarray(y).reshape(-1, 1)
    ).ravel()

def inverse_target(scaler, y):
    return scaler.inverse_transform(
        np.asarray(y).reshape(-1, 1)
    ).ravel()

def calculate_metrics(actual, predicted):
    actual = np.asarray(actual, dtype=float).ravel()
    predicted = np.asarray(predicted, dtype=float).ravel()

    nonzero = np.abs(actual) > 1e-10

    return {
        "MAE": mean_absolute_error(actual, predicted),

        "RMSE": np.sqrt(
            mean_squared_error(actual, predicted)
        ),

        "R2": (
            r2_score(actual, predicted)
            if len(actual) > 1 and np.std(actual) > 0
            else np.nan
        ),

        "MAPE": (
            100 * np.mean(
                np.abs(
                    (actual[nonzero] - predicted[nonzero])
                    / actual[nonzero]
                )
            )
            if nonzero.any()
            else np.nan
        ),

        "sMAPE": 100 * np.mean(
            2 * np.abs(actual - predicted) /
            np.maximum(
                np.abs(actual) + np.abs(predicted),
                1e-10
            )
        )
    }

def build_nn(input_dim, hp):
    model = tf.keras.Sequential([
        tf.keras.layers.Input(
            shape=(input_dim,)
        )
    ])

    for i in range(int(hp["num_layers"])):

        model.add(
            tf.keras.layers.Dense(
                units=int(hp[f"units_{i}"]),
                activation="relu"
            )
        )

        model.add(
            tf.keras.layers.BatchNormalization()
        )

        model.add(
            tf.keras.layers.Dropout(
                float(hp[f"dropout_{i}"])
            )
        )

    model.add(tf.keras.layers.Dense(1))

    model.compile(
        optimizer=tf.keras.optimizers.Adam(
            learning_rate=float(hp["learning_rate"])
        ),
        loss="mse",
        metrics=["mae"]
    )

    return model

def fit_nn(X, y, hp, seed=SEED, validation=None):
    tf.keras.backend.clear_session()
    tf.keras.utils.set_random_seed(seed)

    model = build_nn(X.shape[1], hp)

    arguments = {
        "x": np.asarray(X, dtype=np.float32),
        "y": np.asarray(y, dtype=np.float32),
        "epochs": CFG["nn_max_epochs"],
        "batch_size": CFG["batch_size"],
        "verbose": 0
    }

    if validation is not None:
        arguments["validation_data"] = validation
        arguments["callbacks"] = [
            tf.keras.callbacks.EarlyStopping(
                monitor="val_loss",
                patience=CFG["nn_patience"],
                restore_best_weights=True
            )
        ]

    start = time.perf_counter()
    history = model.fit(**arguments).history
    elapsed = time.perf_counter() - start

    return model, history, elapsed

def predict_nn(model, X):
    return model.predict(
        np.asarray(X, dtype=np.float32),
        verbose=0
    ).ravel()

def tune_tree(name, train_df):

    if name == "RF":

        estimator = RandomForestRegressor(
            random_state=SEED,
            n_jobs=2
        )

        grid = {
            "model__n_estimators": [100, 200],
            "model__max_depth": [10, 20, None],
            "model__max_features": ["sqrt", "log2"]
        }

    else:

        estimator = XGBRegressor(
            random_state=SEED,
            n_jobs=2,
            tree_method="hist",
            objective="reg:squarederror"
        )

        grid = {
            "model__n_estimators": [100, 200],
            "model__max_depth": [3, 6],
            "model__learning_rate": [0.01, 0.1],
            "model__subsample": [0.8, 1.0],
            "model__colsample_bytree": [0.8, 1.0]
        }

    pipe = Pipeline([
        ("processor", make_processor()),
        ("model", estimator)
    ])

    search = GridSearchCV(
        pipe,
        grid,
        cv=KFold(
            n_splits=CFG["tree_tuning_folds"],
            shuffle=True,
            random_state=SEED
        ),
        scoring="neg_mean_absolute_error",
        n_jobs=1,
        refit=True
    )

    start = time.perf_counter()

    search.fit(
        train_df[FEATURES],
        train_df[TARGET]
    )

    elapsed = time.perf_counter() - start

    best = {
        key.replace("model__", ""): value
        for key, value in search.best_params_.items()
    }

    return best, elapsed

def tune_nn(train_df, run_name="main"):

    # Separate validation BEFORE fitting preprocessing.
    dev, valid = train_test_split(
        train_df,
        test_size=0.15,
        random_state=SEED
    )

    _, Xa, Xv = prepare_features(dev, valid)

    scaler = make_y_scaler(
        dev[TARGET],
        "standard"
    )

    ya = scale_target(scaler, dev[TARGET])
    yv = scale_target(scaler, valid[TARGET])

    def builder(hp):

        count = hp.Int(
            "num_layers", 1, 3
        )

        parameters = {
            "num_layers": count,
            "learning_rate": hp.Float(
                "learning_rate",
                1e-4,
                1e-2,
                sampling="log"
            )
        }

        for i in range(count):

            parameters[f"units_{i}"] = hp.Int(
                f"units_{i}",
                min_value=32,
                max_value=256,
                step=32
            )

            parameters[f"dropout_{i}"] = hp.Float(
                f"dropout_{i}",
                min_value=0.2,
                max_value=0.5,
                step=0.1
            )

        return build_nn(
            Xa.shape[1],
            parameters
        )

    tuner = kt.RandomSearch(
        builder,
        objective="val_mae",
        max_trials=CFG["nn_trials"],
        seed=SEED,
        executions_per_trial=1,
        directory=str(META_DIR / "keras_tuner"),
        project_name=run_name,
        overwrite=True
    )

    start = time.perf_counter()

    tuner.search(
        Xa,
        ya,
        validation_data=(Xv, yv),
        epochs=CFG["nn_max_epochs"],
        batch_size=CFG["batch_size"],
        callbacks=[
            tf.keras.callbacks.EarlyStopping(
                monitor="val_loss",
                patience=CFG["nn_patience"],
                restore_best_weights=True
            )
        ],
        verbose=0
    )

    elapsed = time.perf_counter() - start

    chosen = tuner.get_best_hyperparameters(1)[0].values

    hp = {
        "num_layers": int(chosen["num_layers"]),
        "learning_rate": float(
            chosen["learning_rate"]
        )
    }

    for i in range(hp["num_layers"]):
        hp[f"units_{i}"] = int(
            chosen[f"units_{i}"]
        )
        hp[f"dropout_{i}"] = float(
            chosen[f"dropout_{i}"]
        )

    return hp, elapsed

def fit_base(name, X, y, hp, seed=SEED):

    if name == "RF":
        model = create_rf(hp["RF"])
        model.fit(X, y)
        return model

    if name == "XGBoost":
        model = create_xgb(hp["XGBoost"])
        model.fit(X, y)
        return model

    if name == "NN":
        model, _, _ = fit_nn(
            X, y, hp["NN"], seed=seed
        )
        return model

    if name == "Linear":
        model = LinearRegression()
        model.fit(X, y)
        return model

    if name == "Dummy":
        model = DummyRegressor(strategy="mean")
        model.fit(X, y)
        return model

    raise ValueError(name)

def predict_base(name, model, X):
    if name == "NN":
        return predict_nn(model, X)

    return model.predict(X)

BASE = ["RF", "XGBoost", "NN"]

def run_stacking(
    train_df,
    test_df,
    hp,
    target_scaling="standard",
    include_linear=True,
    save_nn_history=False,
    oof_folds=None
):

    names = BASE.copy()

    if include_linear:
        names.append("Linear")

    y = train_df[TARGET].to_numpy()

    folds = KFold(
        n_splits=oof_folds or CFG["oof_folds"],
        shuffle=True,
        random_state=SEED
    )

    OOF = np.full(
        (len(train_df), len(names)),
        np.nan
    )

    # Preprocessing and target scaling are fitted separately
    # within every OOF training fold.

    oof_start = time.perf_counter()

    for fold_number, (a, b) in enumerate(
        folds.split(train_df)
    ):

        df_a = train_df.iloc[a]
        df_b = train_df.iloc[b]

        _, Xa, Xb = prepare_features(df_a, df_b)

        scaler = make_y_scaler(
            df_a[TARGET],
            target_scaling
        )

        ya = scale_target(
            scaler,
            df_a[TARGET]
        )

        for j, name in enumerate(names):

            model = fit_base(
                name,
                Xa,
                ya,
                hp,
                seed=SEED + fold_number
            )

            OOF[b, j] = inverse_target(
                scaler,
                predict_base(name, model, Xb)
            )

    if not np.isfinite(OOF).all():
        raise RuntimeError("Invalid OOF predictions")

    oof_seconds = time.perf_counter() - oof_start

    Z = OOF[:, :3]

    # Ridge alpha selection uses ONLY training OOF predictions.
    alpha_options = [
        0.001, 0.01, 0.1, 1, 10, 100
    ]

    meta_cv = KFold(
        n_splits=5,
        shuffle=True,
        random_state=SEED
    )

    sensitivity = []

    for alpha in alpha_options:

        fold_scores = []

        for a, b in meta_cv.split(Z):
            model = Ridge(alpha=alpha)
            model.fit(Z[a], y[a])

            fold_scores.append(
                mean_absolute_error(
                    y[b],
                    model.predict(Z[b])
                )
            )

        sensitivity.append({
            "Alpha": alpha,
            "Training_Meta_CV_MAE": np.mean(fold_scores)
        })

    selected_alpha = min(
        sensitivity,
        key=lambda row: row["Training_Meta_CV_MAE"]
    )["Alpha"]

    ridge = Ridge(alpha=selected_alpha)
    ridge.fit(Z, y)

    # NNLS: no intercept, nonnegative weights,
    # no sum-to-one constraint.

    nnls_weights, _ = nnls(Z, y)

    # Fit preprocessing and base models on the complete
    # training portion, never on the evaluation set.

    processor, Xfull, Xtest = prepare_features(
        train_df, test_df
    )

    final_scaler = make_y_scaler(
        y,
        target_scaling
    )

    scaled_y = scale_target(
        final_scaler, y
    )

    models = {}
    predictions = {}
    fit_times = []
    nn_history = None

    for name in names:

        start = time.perf_counter()

        if name == "NN" and save_nn_history:

            sub_a, sub_b = train_test_split(
                np.arange(len(train_df)),
                test_size=0.15,
                random_state=SEED
            )

            # Diagnostic learning curve:
            # Fit its feature/target transformations on
            # diagnostic training observations only.

            diag_train = train_df.iloc[sub_a]
            diag_valid = train_df.iloc[sub_b]

            _, Xdiag, Xdiag_valid = prepare_features(
                diag_train, diag_valid
            )

            diag_scaler = make_y_scaler(
                diag_train[TARGET],
                target_scaling
            )

            ydiag = scale_target(
                diag_scaler,
                diag_train[TARGET]
            )

            ydiag_valid = scale_target(
                diag_scaler,
                diag_valid[TARGET]
            )

            _, nn_history, _ = fit_nn(
                Xdiag,
                ydiag,
                hp["NN"],
                validation=(
                    Xdiag_valid,
                    ydiag_valid
                )
            )

        model = fit_base(
            name,
            Xfull,
            scaled_y,
            hp
        )

        seconds = time.perf_counter() - start

        fit_times.append({
            "Model": name,
            "Training_Seconds": seconds,
            "Training_Rows": len(train_df)
        })

        models[name] = model

        predictions[name] = inverse_target(
            final_scaler,
            predict_base(name, model, Xtest)
        )

    Ztest = np.column_stack([
        predictions[name] for name in BASE
    ])

    predictions["Stack_Ridge"] = ridge.predict(Ztest)
    predictions["Stack_Mean"] = Ztest.mean(axis=1)
    predictions["Stack_NNLS"] = (
        Ztest @ nnls_weights
    )

    # Base-learner ablations.

    for removed in BASE:

        columns = [
            i for i, name in enumerate(BASE)
            if name != removed
        ]

        reduced = Ridge(alpha=selected_alpha)
        reduced.fit(
            Z[:, columns],
            y
        )

        predictions["Without_" + removed] = (
            reduced.predict(Ztest[:, columns])
        )

    # Linear learner as an alternative stacking input.

    if include_linear:

        linear_instead = Ridge(
            alpha=selected_alpha
        )

        linear_instead.fit(
            OOF[:, [0, 1, 3]],
            y
        )

        predictions["Stack_LinearInsteadNN"] = (
            linear_instead.predict(
                np.column_stack([
                    Ztest[:, 0],
                    Ztest[:, 1],
                    predictions["Linear"]
                ])
            )
        )

        four_model = Ridge(alpha=selected_alpha)
        four_model.fit(OOF, y)

        predictions["Stack_WithLinear"] = (
            four_model.predict(
                np.column_stack([
                    Ztest,
                    predictions["Linear"]
                ])
            )
        )

    def predict_ensemble(raw_frame):

        X = np.asarray(
            processor.transform(raw_frame[FEATURES]),
            dtype=np.float32
        )

        columns = []

        for name in BASE:

            predicted = predict_base(
                name, models[name], X
            )

            columns.append(
                inverse_target(
                    final_scaler,
                    predicted
                )
            )

        return ridge.predict(
            np.column_stack(columns)
        )

    return {
        "predictions": predictions,
        "models": models,
        "processor": processor,
        "target_scaler": final_scaler,
        "ridge": ridge,
        "selected_alpha": selected_alpha,
        "alpha_table": pd.DataFrame(sensitivity),
        "nnls_weights": nnls_weights,
        "oof_seconds": oof_seconds,
        "fit_times": pd.DataFrame(fit_times),
        "nn_history": nn_history,
        "predict": predict_ensemble
    }

def bootstrap_analysis():

        "RF", "XGBoost", "NN", "Stack_Ridge"
    ]

    for iteration in range(
        CFG["bootstrap_iterations"]

            0,

            len(Y_TEST)
        )

        for name in selected_models:

            m = calculate_metrics(
                Y_TEST[sample],
                PRED[name][sample]
            )

            records.append({
                "Iteration": iteration,
                "Model": name,
                "MAE": m["MAE"],
                "RMSE": m["RMSE"],
                "R2": m["R2"]
            })

    boot = pd.DataFrame(records)
    intervals = []

    for name, group in boot.groupby("Model"):

        result = {"Model": name}

        for metric in ["MAE", "RMSE", "R2"]:

            result[f"{metric}_Lower95"] = (
                group[metric].quantile(0.025)
            )

            result[f"{metric}_Upper95"] = (
                group[metric].quantile(0.975)
            )

        intervals.append(result)

    save_table(
        "RA_R1_RA_T4",
        "bootstrap_95CI",
        pd.DataFrame(intervals),
        "Bootstrap 95% performance intervals"
    )

    pairs = []

    for iteration, group in boot.groupby("Iteration"):

        z = group.set_index("Model")

        pairs.append({
            "Iteration": iteration,
            "MAE_XGBoost_minus_Stack": (
                z.loc["XGBoost", "MAE"]
                - z.loc["Stack_Ridge", "MAE"]
            ),
            "RMSE_XGBoost_minus_Stack": (
                z.loc["XGBoost", "RMSE"]
                - z.loc["Stack_Ridge", "RMSE"]
            )
        })

    differences = pd.DataFrame(pairs)

    save_table(
        "RA_R1_RA_T4",
        "paired_bootstrap_differences",
        differences,
        "Paired bootstrap model comparison"
    )

    for metric in [
        "MAE_XGBoost_minus_Stack",
        "RMSE_XGBoost_minus_Stack"
    ]:

        plt.figure(figsize=(8, 5))

        plt.hist(
            differences[metric],
            bins=35,
            edgecolor="black",
            linewidth=0.4
        )

        plt.axvline(
            0,
            linestyle="--",
            color="black"
        )

        plt.xlabel(
            "Difference (XGBoost − Stacking)"
        )
        plt.ylabel("Bootstrap Frequency")
        plt.title(
            metric.replace("_", " ")
        )

        save_fig(
            "RA_R1_RA_T4",

def ensemble_shap():

    if not HAS_SHAP:
        raise RuntimeError(
            "SHAP is not installed"
        )

    background = TRAIN[FEATURES].sample(
        CFG["shap_background"],
        random_state=SEED
    )

    explain = TEST[FEATURES].sample(
        CFG["shap_explain"],
        random_state=SEED
    )

    categories = {
        c: sorted(data[c].unique().tolist())
        for c in CATEGORICAL
    }

    def encode(df):

        result = df[FEATURES].copy()

        for c in CATEGORICAL:
            result[c] = pd.Categorical(
                result[c],
                categories=categories[c]
            ).codes

        return result.astype(float).to_numpy()

    def decode(array):

        df = pd.DataFrame(
            array,
            columns=FEATURES
        )

        for c in CATEGORICAL:

            idx = np.clip(
                np.rint(df[c]).astype(int),
                0,
                len(categories[c]) - 1
            )

            df[c] = np.asarray(
                categories[c],
                dtype=object
            )[idx]

        return df

    explainer = shap.KernelExplainer(
        lambda x: MAIN["predict"](decode(x)),
        encode(background)
    )

    shap_values = np.asarray(
        explainer.shap_values(
            encode(explain),
            nsamples=CFG["shap_nsamples"]
        )
    )

    if shap_values.ndim == 3:
        shap_values = shap_values[:, :, 0]

    importance = pd.DataFrame({
        "Feature": FEATURES,
        "Mean_Absolute_SHAP": np.mean(
            np.abs(shap_values),
            axis=0
        )
    })

    save_table(
        "RA_O4_RA_O5_RB_M2",
        "ensemble_shap_values",
        importance,
        "Approximate ensemble-level SHAP attribution"
    )

    result = importance.sort_values(
        "Mean_Absolute_SHAP"
    )

    plt.figure(figsize=(10, 6))

    plt.barh(
        result.Feature,
        result.Mean_Absolute_SHAP
    )

    plt.xlabel(
        "Mean Absolute SHAP Value (hg/ha)"
    )
    plt.title(
        "Stacking Ensemble Feature Importance"
    )

    save_fig(
        "RA_O4_RA_O5_RB_M2",
        "ensemble_shap_importance",
        "Approximate ensemble SHAP importance"
    )

if CFG["run_shap"]:
    run_stage("Ensemble SHAP", ensemble_shap)

def attention_comparator():

    _, Xa, Xb = prepare_features(
        TRAIN,
        TEST
    )

    scaler = make_y_scaler(
        TRAIN[TARGET]
    )

    ya = scale_target(
        scaler,
        TRAIN[TARGET]
    )

    tf.keras.backend.clear_session()
    tf.keras.utils.set_random_seed(SEED)

    inputs = tf.keras.Input(
        shape=(Xa.shape[1],)
    )

    x = tf.keras.layers.Reshape(
        (Xa.shape[1], 1)
    )(inputs)

    x = tf.keras.layers.Dense(16)(x)

    attended = tf.keras.layers.MultiHeadAttention(
        num_heads=2,
        key_dim=8
    )(x, x)

    x = tf.keras.layers.LayerNormalization()(
        x + attended
    )

    x = tf.keras.layers.GlobalAveragePooling1D()(x)

    x = tf.keras.layers.Dense(
        64,
        activation="relu"
    )(x)

    output = tf.keras.layers.Dense(1)(x)

    model = tf.keras.Model(inputs, output)

    model.compile(
        optimizer=tf.keras.optimizers.Adam(1e-3),
        loss="mse",
        metrics=["mae"]
    )

    start = time.perf_counter()

    history = model.fit(
        Xa,
        ya,
        validation_split=0.10,
        epochs=CFG["nn_max_epochs"],

def temporal_validation():

    years = np.sort(
        data.Year.unique()
    )

    cutoff = (
        2015
        if (
            (data.Year < 2015).sum() >= 100
            and (data.Year >= 2015).sum() >= 50
        )
        else int(np.quantile(years, 0.75))
    )

    past = data[
        data.Year < cutoff
    ].reset_index(drop=True)

    future = data[
        data.Year >= cutoff
    ].reset_index(drop=True)

    if len(past) < 100 or len(future) < 40:
        raise ValueError(
            "Insufficient records for temporal validation"
        )

    # Retune model hyperparameters using only historical data.

    rf, _ = tune_tree("RF", past)
    xgb, _ = tune_tree("XGBoost", past)
    nn, _ = tune_nn(past, "temporal")

    temporal_hp = {
        "RF": rf,
        "XGBoost": xgb,
        "NN": nn
    }

    result = run_stacking(
        past,
        future,
        temporal_hp,
        include_linear=False
    )

    records = []

    for name in BASE + ["Stack_Ridge"]:

        records.append({
            "Model": name,
            "Train_Before": cutoff,
            "Test_From": cutoff,
            "Train_Rows": len(past),
            "Test_Rows": len(future),
            **calculate_metrics(
                future[TARGET],
                result["predictions"][name]
            )
        })

    save_table(
        "RA_M2_RA_R8",
        "temporal_model_results",
        pd.DataFrame(records),
        "Chronological validation with historical-only tuning"
    )

if CFG["run_temporal_validation"]:
    run_stage(
        "Chronological validation",
        temporal_validation

def outer_cross_validation():

    outer = KFold(
        n_splits=CFG["outer_cv_folds"],
        shuffle=True,
        random_state=SEED
    )

    results = []

    for fold, (a, b) in enumerate(
        outer.split(data),
        start=1
    ):

        print(
            f"\nOuter fold {fold}/{CFG['outer_cv_folds']}"
        )

        fold_train = data.iloc[a].reset_index(drop=True)
        fold_test = data.iloc[b].reset_index(drop=True)

        # IMPORTANT: Retune independently inside outer train.
        rf, _ = tune_tree("RF", fold_train)
        xgb, _ = tune_tree("XGBoost", fold_train)
        nn, _ = tune_nn(
            fold_train,
            f"outer_{fold}"
        )

        fold_hp = {
            "RF": rf,
            "XGBoost": xgb,
            "NN": nn
        }

        fold_result = run_stacking(
            fold_train,
            fold_test,
            fold_hp,
            include_linear=False
        )

        for name in BASE + ["Stack_Ridge"]:

            results.append({
                "Fold": fold,
                "Model": name,
                **calculate_metrics(
                    fold_test[TARGET],
                    fold_result["predictions"][name]
                )
            })

        save_table(
            "RA_T2_RA_T4",
            f"outer_cv_progress_fold_{fold}",
            pd.DataFrame(results),
            "Completed outer-fold validation results"
        )

    frame = pd.DataFrame(results)

    summary = frame.groupby("Model")[
        ["MAE", "RMSE", "R2"]
    ].agg(["mean", "std"])

    summary.columns = [
        "_".join(c)
        for c in summary.columns
    ]

    summary = summary.reset_index()

    save_table(
        "RA_T2_RA_T4",
        "outer_cv_summary",
        summary,
        "Outer-fold performance mean and standard deviation"
    )

    plt.figure(figsize=(9, 5.5))

    for name, group in frame.groupby("Model"):

        plt.plot(
            group.Fold,
            group.R2,
            marker="o",
            label=name
        )

    plt.xlabel("Outer Fold")
    plt.ylabel("R²")
    plt.title("Cross-Validation Performance")
    plt.legend()

    save_fig(
        "RA_T2_RA_T4",
        "outer_cv_r2",
        "Outer cross-validation R² comparison"
    )

if CFG["run_outer_cv"]:
    run_stage(
        "Nested outer cross-validation",
        outer_cross_validation

def scalability_analysis():

    rows = []
    stack_rows = []

    for factor in CFG["scalability_factors"]:

        print(f"\nBenchmarking {factor}x dataset size")

        workload = TRAIN.sample(
            n=len(TRAIN) * factor,
            replace=(factor > 1),
            random_state=SEED + factor
        ).reset_index(drop=True)

        _, X, _ = prepare_features(
            workload,
            TEST
        )

        scaler = make_y_scaler(
            workload[TARGET]
        )

        y = scale_target(
            scaler,
            workload[TARGET]
        )

        for name in BASE:

            start = time.perf_counter()

            model = fit_base(
                name,
                X,
                y,
                HP
            )

            train_seconds = (
                time.perf_counter() - start
            )

            batch = X[:min(100, len(X))]

            # Warm-up.
            predict_base(name, model, batch)

            samples = []

            for repeat in range(5):

                start = time.perf_counter()

                predict_base(
                    name,
                    model,
                    batch
                )

                samples.append(
                    time.perf_counter() - start
                )

            latency = (
                1000 * np.median(samples) / len(batch)
            )

            rows.append({
                "Scale": factor,
                "Observations": len(workload),
                "Model": name,
                "Training_Seconds": train_seconds,
                "Inference_ms_per_sample": latency,
                "Workload": (
                    "Original records"
                    if factor == 1
                    else "Replicated computational workload"
                )
            })

        # Whole stacking model, including OOF fitting.
        start = time.perf_counter()

        test_batch = TEST.iloc[
            :min(100, len(TEST))
        ]

        stack = run_stacking(
            workload,
            test_batch,
            HP,
            include_linear=False
        )

        elapsed = time.perf_counter() - start

        # End-to-end stacking inference.
        stack["predict"](test_batch)

        repeats = []

        for repeat in range(5):
            start = time.perf_counter()
            stack["predict"](test_batch)
            repeats.append(
                time.perf_counter() - start
            )

        stack_rows.append({
            "Scale": factor,
            "Observations": len(workload),
            "OOF_Seconds": stack["oof_seconds"],
            "Total_Training_Seconds": elapsed,
            "Inference_ms_per_sample": (
                1000 * np.median(repeats) /
                len(test_batch)
            ),
            "Workload": (
                "Original records"
                if factor == 1
                else "Replicated computational workload"
            )
        })

        # Save incrementally after every scale.

        save_table(
            "RA_T3_RB_M1_RB_Q5",
            "base_scalability_progress",
            pd.DataFrame(rows),
            "Base-learner scalability benchmark"
        )

        save_table(
            "RA_T3_RB_M1_RB_Q5",
            "stack_scalability_progress",
            pd.DataFrame(stack_rows),
            "Stacking scalability benchmark"
        )

    base_frame = pd.DataFrame(rows)
    stack_frame = pd.DataFrame(stack_rows)

    plt.figure(figsize=(9, 5.5))

    for name, group in base_frame.groupby("Model"):

        group = group.sort_values("Observations")

        plt.plot(
            group.Observations,
            group.Training_Seconds,
            marker="o",
            label=name
        )

    plt.xlabel("Training Observations")
    plt.ylabel("Training Time (seconds)")
    plt.title("Base-Model Training Scalability")
    plt.legend()

    save_fig(
        "RA_T3_RB_M1_RB_Q5",
        "base_training_scalability",
        "Training time versus dataset scale"
    )

    plt.figure(figsize=(9, 5.5))

    plt.plot(
        stack_frame.Observations,
        stack_frame.Total_Training_Seconds,
        marker="o"
    )

    plt.xlabel("Training Observations")
    plt.ylabel("Total Training Time (seconds)")
    plt.title("Stacking Ensemble Scalability")

    save_fig(
        "RA_T3_RB_M1_RB_Q5",
        "stack_training_scalability",
        "End-to-end stacking training time"
    )

    plt.figure(figsize=(9, 5.5))

    plt.plot(
        stack_frame.Observations,
        stack_frame.Inference_ms_per_sample,
        marker="o"
    )

    plt.xlabel("Training Observations")
    plt.ylabel("Inference Time (ms/sample)")
    plt.title("Stacking Ensemble Inference Latency")

    save_fig(
        "RA_T3_RB_M1_RB_Q5",
        "stack_inference_scalability",
        "End-to-end stacking inference latency"
    )

if CFG["run_scalability"]:
    run_stage(
        "Training and inference scalability",
        scalability_analysis
