# 3. LEAKAGE-SAFE PREPROCESSING
# =====================================================================

def make_processor():
    try:
        encoder = OneHotEncoder(
            handle_unknown="ignore",
            sparse_output=False
        )
    except TypeError:
        encoder = OneHotEncoder(
            handle_unknown="ignore",
            sparse=False
        )

    return ColumnTransformer(
        transformers=[
            ("categorical", encoder, CATEGORICAL),
            ("numerical", StandardScaler(), NUMERICAL)
        ],
        sparse_threshold=0
    )


def prepare_features(train_df, other_df):
    processor = make_processor()

    X_train = np.asarray(
        processor.fit_transform(train_df[FEATURES]),
        dtype=np.float32
    )

    X_other = np.asarray(
        processor.transform(other_df[FEATURES]),
        dtype=np.float32
    )

    return processor, X_train, X_other


def prepare_target(train_df):
    scaler = StandardScaler()

    y = train_df[TARGET].to_numpy(dtype=float)

    scaled = scaler.fit_transform(
        y.reshape(-1, 1)
    ).ravel().astype(np.float32)

    return scaler, scaled


def inverse_target(scaler, prediction):
    return scaler.inverse_transform(
        np.asarray(prediction).reshape(-1, 1)
    ).ravel()


def evaluate(actual, predicted):
    return {
        "MAE": mean_absolute_error(actual, predicted),
        "RMSE": np.sqrt(
            mean_squared_error(actual, predicted)
        ),
        "R2": r2_score(actual, predicted)
    }

# 4. NEURAL NETWORK ARCHITECTURE
# =====================================================================

def build_nn(input_dim, hp):
    model = tf.keras.Sequential([
        tf.keras.layers.Input(shape=(input_dim,))
    ])

    for i in range(int(hp["num_layers"])):
        model.add(
            tf.keras.layers.Dense(
                int(hp[f"units_{i}"]),
                activation="relu",
                kernel_initializer="glorot_uniform",
                bias_initializer="zeros"
            )
        )

        model.add(
            tf.keras.layers.BatchNormalization()
        )

        model.add(
            tf.keras.layers.Dropout(
                float(hp[f"dropout_{i}"])
            )
        )

    model.add(tf.keras.layers.Dense(1))

    model.compile(
        optimizer=tf.keras.optimizers.Adam(
            learning_rate=float(hp["learning_rate"])
        ),
        loss="mse",
        metrics=["mae"]
    )

    return model


def fit_nn(X, y, hp, seed=SEED):
    tf.keras.backend.clear_session()
    tf.keras.utils.set_random_seed(seed)

    model = build_nn(X.shape[1], hp)

    model.fit(
        np.asarray(X, dtype=np.float32),
        np.asarray(y, dtype=np.float32),
        epochs=EPOCHS,
        batch_size=BATCH_SIZE,
        verbose=0
    )

    return model

# =====================================================================
# 5. HISTORICAL-ONLY TREE HYPERPARAMETER TUNING
# =====================================================================

def create_tree(name, hp):
    if name == "RF":
        return RandomForestRegressor(
            random_state=SEED,
            n_jobs=2,
            **hp
        )

    return XGBRegressor(
        random_state=SEED,
        n_jobs=2,
        tree_method="hist",
        objective="reg:squarederror",
        **hp
    )


def tune_tree(name):
    if name == "RF":
        estimator = RandomForestRegressor(
            random_state=SEED,
            n_jobs=2
        )

        grid = {
            "model__n_estimators": [100, 200],
            "model__max_depth": [10, 20, None],
            "model__max_features": ["sqrt", "log2"]
        }

    else:
        estimator = XGBRegressor(
            random_state=SEED,
            n_jobs=2,
            tree_method="hist",
            objective="reg:squarederror"
        )

        grid = {
            "model__n_estimators": [100, 200],
            "model__max_depth": [3, 6],
            "model__learning_rate": [0.01, 0.1],
            "model__subsample": [0.8, 1.0],
            "model__colsample_bytree": [0.8, 1.0]
        }

    pipeline = Pipeline([
        ("processor", make_processor()),
        ("model", estimator)
    ])

    search = GridSearchCV(
        pipeline,
        grid,
        cv=KFold(
            n_splits=TUNING_FOLDS,
            shuffle=True,
            random_state=SEED
        ),
        scoring="neg_mean_absolute_error",
        n_jobs=1,
        refit=True
    )

    start = time.perf_counter()

    search.fit(TRAIN[FEATURES], TRAIN[TARGET])

    elapsed = time.perf_counter() - start

    selected = {
        k.replace("model__", ""): v
        for k, v in search.best_params_.items()
    }

    return selected, elapsed

def nn_builder(hp):
    count = hp.Int("num_layers", 1, 3)

    parameters = {
        "num_layers": count,
        "learning_rate": hp.Float(
            "learning_rate",
            1e-4,
            1e-2,
            sampling="log"
        )
    }

    for i in range(count):
        parameters[f"units_{i}"] = hp.Int(
            f"units_{i}",
            32,
            256,
            step=32
        )

        parameters[f"dropout_{i}"] = hp.Float(
            f"dropout_{i}",
            0.2,
            0.5,
            step=0.1
        )

    return build_nn(
        X_DEV.shape[1],
        parameters
    )

    objective="val_mae",
    max_trials=NN_TRIALS,
    seed=SEED,
    executions_per_trial=1,

    project_name="temporal_benchmark",
    overwrite=True

HP = {

    "NN": NN_HP

def fit_base(name, X, y, seed=SEED):
    if name == "NN":
        return fit_nn(X, y, HP["NN"], seed=seed)

    model = create_tree(name, HP[name])
    model.fit(X, y)
    return model


def raw_prediction(name, model, X):
    if name == "NN":
        return model.predict(
            np.asarray(X, dtype=np.float32),
            verbose=0
        ).ravel()

    return np.asarray(model.predict(X)).ravel()

# 8. FIVE-FOLD OUT-OF-FOLD STACKING
# =====================================================================

print("\n[4/6] Building OOF stacking predictions...")

OOF = np.full(
    (len(TRAIN), len(BASE)),
    np.nan,
    dtype=float
)

folds = KFold(
    n_splits=OOF_FOLDS,
    shuffle=True,
    random_state=SEED
)

OOF_START = time.perf_counter()

for fold_index, (a, b) in enumerate(folds.split(TRAIN)):
    train_fold = TRAIN.iloc[a]
    validation_fold = TRAIN.iloc[b]

    _, X_FOLD, X_VAL = prepare_features(
        train_fold,
        validation_fold
    )

    fold_scaler, Y_FOLD = prepare_target(train_fold)

    for j, name in enumerate(BASE):
        model = fit_base(
            name,
            X_FOLD,
            Y_FOLD,
            seed=SEED + fold_index
        )

        scaled_prediction = raw_prediction(
            name,
            model,
            X_VAL
        )

        OOF[b, j] = inverse_target(
            fold_scaler,
            scaled_prediction
        )

        del model
        gc.collect()

    print(f"Completed fold {fold_index + 1}/{OOF_FOLDS}")

if not np.isfinite(OOF).all():
    raise RuntimeError("Invalid OOF predictions.")

OOF_TIME = time.perf_counter() - OOF_START

print(f"OOF generation time: {OOF_TIME:.2f} seconds")

# =====================================================================
# 9. RIDGE META-LEARNER TUNING AND TRAINING
# =====================================================================

print("\n[5/6] Tuning and fitting Ridge meta-learner...")

Y_TRAIN_ORIGINAL = TRAIN[TARGET].to_numpy(dtype=float)

RIDGE_ALPHAS = [
    0.001, 0.01, 0.1, 1, 10, 100
]

META_START = time.perf_counter()

META_CV = KFold(
    n_splits=5,
    shuffle=True,
    random_state=SEED
)

alpha_results = []

for alpha in RIDGE_ALPHAS:
    scores = []

    for a, b in META_CV.split(OOF):
        trial_ridge = Ridge(alpha=alpha)

        trial_ridge.fit(
            OOF[a],
            Y_TRAIN_ORIGINAL[a]
        )

        scores.append(
            mean_absolute_error(
                Y_TRAIN_ORIGINAL[b],
                trial_ridge.predict(OOF[b])
            )
        )

    alpha_results.append({
        "Alpha": alpha,
        "CV_MAE": np.mean(scores)
    })

BEST_ALPHA = min(
    alpha_results,
    key=lambda item: item["CV_MAE"]
)["Alpha"]

ridge = Ridge(alpha=BEST_ALPHA)
ridge.fit(OOF, Y_TRAIN_ORIGINAL)

META_TIME = time.perf_counter() - META_START

print("Selected Ridge alpha:", BEST_ALPHA)
print(f"Meta-learner tuning/fitting time: {META_TIME:.2f}s")

# 10. FINAL HISTORICAL TRAINING

    TRAIN,
    TEST

        name,
        XTRAIN,

        time.perf_counter() - start

    prediction_scaled = raw_prediction(
        name,
        model,
        XTEST
    )

    base_predictions[name] = inverse_target(
        final_scaler,
        prediction_scaled
    )

    print(
        f"{name} final fit time: "

    )

STACK_FEATURES = np.column_stack([

    for name in BASE
])

base_predictions["Stack_Ridge"] = ridge.predict(

)

FINAL_STAGE_ELAPSED = time.perf_counter() - TOTAL_STACK_START

#
# The recorded stage times may include small timer/warm-up overhead.

def predict_complete_base(name):
    X = np.asarray(
        final_processor.transform(TEST[FEATURES]),
        dtype=np.float32
    )

    result = raw_prediction(
        name,
        final_models[name],
        X
    )

    return inverse_target(
        final_scaler,
        result
    )


def predict_complete_stack():
    X = np.asarray(
        final_processor.transform(TEST[FEATURES]),
        dtype=np.float32
    )

    predictions = []

    for name in BASE:
        scaled = raw_prediction(
            name,
            final_models[name],
            X
        )

        predictions.append(
            inverse_target(
                final_scaler,
                scaled
            )
        )

    return ridge.predict(
        np.column_stack(predictions)
    )


functions = {
    "RF": lambda: predict_complete_base("RF"),
    "XGBoost": lambda: predict_complete_base("XGBoost"),
    "NN": lambda: predict_complete_base("NN"),
    "Stack_Ridge": predict_complete_stack
}

inference_rows = []

for name, fn in functions.items():

    print(f"Benchmarking {name}...")

    # Warm-up predictions
    for _ in range(INFERENCE_WARMUPS):
        fn()

    durations = []

    for repeat in range(INFERENCE_REPEATS):
        start = time.perf_counter()

        result = fn()

        duration = time.perf_counter() - start

        if len(result) != len(TEST):
            raise RuntimeError(
                f"Invalid output length for {name}"
            )

        durations.append(duration)

    durations = np.asarray(durations)

    median_batch_ms = (
        1000.0 * np.median(durations)
    )

    inference_rows.append({
        "Model": name,
        "Inference_ms_per_sample": (
            median_batch_ms / len(TEST)
        ),
        "Median_batch_time_ms": median_batch_ms,
        "Mean_batch_time_ms": (
            1000.0 * np.mean(durations)
        ),
        "Std_batch_time_ms": (
            1000.0 * np.std(durations)
        ),
        "Batch_size_samples": len(TEST),
        "Repeats": INFERENCE_REPEATS
    })

inference_df = pd.DataFrame(inference_rows)
